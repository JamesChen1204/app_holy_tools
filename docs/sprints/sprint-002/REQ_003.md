## 🔧 後端實現 (Go)

### 1. 設定檔管理服務

```go
// config/service.go
package config

import (
    "encoding/json"
    "io/ioutil"
    "os"
    "path/filepath"
    "time"
)

type ConfigService struct {
    configPath string
    config     *Config
}

// NewConfigService 創建設定服務
func NewConfigService() *ConfigService {
    homeDir, _ := os.UserHomeDir()
    configDir := filepath.Join(homeDir, ".powermoat_app_tools")
    
    // 確保目錄存在
    os.MkdirAll(configDir, 0755)
    
    return &ConfigService{
        configPath: filepath.Join(configDir, "config.json"),
    }
}

// LoadConfig 載入設定檔
func (s *ConfigService) LoadConfig() (*Config, error) {
    // 如果檔案不存在，創建預設設定
    if _, err := os.Stat(s.configPath); os.IsNotExist(err) {
        return s.createDefaultConfig()
    }
    
    data, err := ioutil.ReadFile(s.configPath)
    if err != nil {
        return nil, err
    }
    
    var config Config
    if err := json.Unmarshal(data, &config); err != nil {
        return nil, err
    }
    
    s.config = &config
    return &config, nil
}

// SaveConfig 儲存設定檔
func (s *ConfigService) SaveConfig(config *Config) error {
    // 更新元數據
    config.Metadata.LastModified = time.Now()
    
    // 序列化為 JSON
    data, err := json.MarshalIndent(config, "", "  ")
    if err != nil {
        return err
    }
    
    // 寫入檔案
    if err := ioutil.WriteFile(s.configPath, data, 0644); err != nil {
        return err
    }
    
    s.config = config
    return nil
}

// GetConfig 取得當前設定
func (s *ConfigService) GetConfig() *Config {
    if s.config == nil {
        s.LoadConfig()
    }
    return s.config
}

// createDefaultConfig 創建預設設定
func (s *ConfigService) createDefaultConfig() (*Config, error) {
    config := &Config{
        Version: "1.0.0",
        Database: Database{
            Type: "mssql",
            Host: "localhost",
            Port: 1433,
        },
        Application: Application{
            Language: "zh-TW",
            Theme:    "light",
        },
        Metadata: Metadata{
            CreatedAt:    time.Now(),
            LastModified: time.Now(),
            ModifiedBy:   "system",
        },
    }
    
    if err := s.SaveConfig(config); err != nil {
        return nil, err
    }
    
    return config, nil
}
```

### 2. 資料庫連線測試

```go
// config/database.go
package config

import (
    "database/sql"
    "fmt"
    "time"
    
    _ "github.com/denisenkom/go-mssqldb"
    _ "github.com/go-sql-driver/mysql"
)

// TestConnection 測試資料庫連線
func (s *ConfigService) TestConnection(db Database) *ConnectionTestResult {
    start := time.Now()
    
    var dsn string
    switch db.Type {
    case "mssql":
        dsn = fmt.Sprintf("server=%s;port=%d;user id=%s;password=%s;database=%s",
            db.Host, db.Port, db.Username, db.Password, db.Database)
    case "mysql":
        dsn = fmt.Sprintf("%s:%s@tcp(%s:%d)/%s",
            db.Username, db.Password, db.Host, db.Port, db.Database)
    default:
        return &ConnectionTestResult{
            Success: false,
            Message: "不支援的資料庫類型",
        }
    }
    
    conn, err := sql.Open(db.Type, dsn)
    if err != nil {
        return &ConnectionTestResult{
            Success: false,
            Message: fmt.Sprintf("連線失敗: %v", err),
        }
    }
    defer conn.Close()
    
    if err := conn.Ping(); err != nil {
        return &ConnectionTestResult{
            Success: false,
            Message: fmt.Sprintf("Ping 失敗: %v", err),
        }
    }
    
    latency := time.Since(start).Milliseconds()
    
    return &ConnectionTestResult{
        Success: true,
        Message: "連線成功",
        Latency: int(latency),
    }
}
```

### 2. 設定管理 Hook

```javascript
// frontend/src/hooks/useConfig.js
import { useState, useEffect } from 'react';
import { GetConfig, SaveConfig } from '../../wailsjs/go/main/App';

export function useConfig() {
  const [config, setConfig] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  const loadConfig = async () => {
    try {
      setLoading(true);
      const cfg = await GetConfig();
      setConfig(cfg);
      setError(null);
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  };

  const saveConfig = async (newConfig) => {
    try {
      await SaveConfig(newConfig);
      setConfig(newConfig);
      return { success: true };
    } catch (err) {
      setError(err.message);
      return { success: false, error: err.message };
    }
  };

  useEffect(() => {
    loadConfig();
  }, []);

  return {
    config,
    loading,
    error,
    reload: loadConfig,
    save: saveConfig
  };
}

## 📊 資料流程

### 1. 設定載入流程

```
應用啟動
    │
    ├─► 檢查設定檔是否存在
    │   ├─► 存在：讀取並解析 JSON
    │   └─► 不存在：創建預設設定
    │
    └─► 載入設定到記憶體
```

### 2. 設定儲存流程

```
用戶填寫表單
    │
    ├─► 前端表單驗證
    │
    ├─► 呼叫 SaveConfig API
    │
    ├─► 後端接收資料
    │   ├─► 更新元數據
    │   └─► 序列化為 JSON
    │
    └─► 寫入檔案系統
```

### 3. 連線測試流程

```
用戶點擊測試連線
    │
    ├─► 前端驗證必填欄位
    │
    ├─► 呼叫 TestConnection API
    │
    ├─► 後端建立資料庫連線
    │   ├─► 根據資料庫類型建立 DSN
    │   ├─► sql.Open() 開啟連線
    │   ├─► conn.Ping() 測試連線
    │   └─► 計算延遲時間
    │
    └─► 返回測試結果
        ├─► 成功：顯示綠色提示 + 延遲
        └─► 失敗：顯示紅色錯誤訊息
```

## ✅ 驗收測試案例

### 1. 功能測試

| 測試項目 | 測試步驟 | 預期結果 |
|---------|---------|---------|
| 首次啟動 | 1. 刪除設定檔<br>2. 啟動應用 | 自動創建預設設定檔 |
| 載入設定 | 1. 啟動應用<br>2. 開啟設定頁面 | 顯示已儲存的設定 |
| 儲存設定 | 1. 修改設定<br>2. 點擊儲存 | 設定寫入檔案 |
| 測試 MSSQL | 1. 輸入 MSSQL 連線資訊<br>2. 點擊測試 | 顯示連線結果 |
| 測試 MySQL | 1. 輸入 MySQL 連線資訊<br>2. 點擊測試 | 顯示連線結果 |
| 密碼顯示/隱藏 | 點擊顯示/隱藏按鈕 | 切換密碼可見性 |
| 表單驗證 | 1. 清空必填欄位<br>2. 嘗試測試/儲存 | 按鈕禁用或顯示錯誤 |

### 2. 邊界測試

| 測試項目 | 輸入 | 預期結果 |
|---------|------|---------|
| 空白 Host | host: "" | 測試按鈕禁用 |
| 無效 Port | port: 99999 | 連線失敗 |
| 錯誤密碼 | 正確 host + 錯誤 password | 驗證失敗訊息 |
| 特殊字元密碼 | 包含 @#$% | 正常加密儲存 |
| 設定檔損壞 | 手動編輯為無效 JSON | 重建預設設定 |

### 3. 性能測試

| 測試項目 | 指標 | 目標值 |
|---------|------|--------|
| 設定載入時間 | 從磁碟讀取 | < 100ms |
| 設定儲存時間 | 寫入磁碟 | < 200ms |
| 連線測試回應 | 本地資料庫 | < 500ms |
| UI 渲染時間 | 頁面載入 | < 300ms |

    // 嘗試從備份恢復或重建預設設定
    log.Printf("設定檔損壞，重建預設設定: %v", err)
    return createDefaultConfig()
}
```

try {
  const result = await TestDatabaseConnection(database);
  if (!result.success) {
    // 顯示友好的錯誤訊息

  }
} catch (error) {

}
